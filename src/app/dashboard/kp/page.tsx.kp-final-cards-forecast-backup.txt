function VerdictBadge({ value }: { value: SignificatorSet["verdict"] }) {
  const map = {
    strong: "✅ Strong Promise",
    moderate: "⚡ Moderate Promise",
    weak: "○ Weak / Latent",
    blocked: "⛔ Blocked / Delayed",
  };

  return <span className={`kp-verdict ${value}`}>{map[value]}</span>;
}

function KPTableRow({ row }: { row: KPRow }) {
  const color = KP_PLANET_COLORS[row.name] ?? "#e8e3f0";
  const starColor = KP_PLANET_COLORS[row.starLord] ?? "#e8e3f0";
  const subColor = KP_PLANET_COLORS[row.subLord] ?? "#e8e3f0";

  return (
    <tr className={row.isSignificantForTopic ? "hit" : ""}>
      <td>
        <strong style={{ color }}>{row.name}</strong>
        {row.retrograde ? <span className="retro">℞</span> : null}
      </td>
      <td>{row.position}</td>
      <td>{row.degreeText}</td>
      <td>{row.signLord}</td>
      <td>{row.nakshatra}</td>
      <td style={{ color: starColor }}>{row.starLord}</td>
      <td>{row.pada}</td>
      <td>
        <span className="sub-pill" style={{ color: subColor }}>
          {row.subLord}
        </span>
      </td>
      <td>{row.subSubLord}</td>
      <td>{row.significance}</td>
    </tr>
  );
}

function SignificatorCard({ item }: { item: SignificatorSet }) {
  return (
    <div className={`kp-sig-card ${item.verdict}`}>
      <div className="kp-sig-head">
        <div>
          <h3>{item.label}</h3>
          <p>Karaka: {item.karaka}</p>

      <div className="kp-score-line">
        <div style={{ width: `${item.score}%` }} />
      </div>

      <p className="kp-note-text">{item.verdictNote}</p>

      <div className="kp-mini-grid">
        <div>
          <span>Event Houses</span>

          <strong>H{item.negativeHouses.join(" · H")}</strong>

        </div>

function HouseLordGrid({ lords }: { lords: Record<number, string> }) {
  return (
    <div className="kp-house-grid">
      {Array.from({ length: 12 }, (_, index) => {
        const house = index + 1;
        const lord = lords[house] ?? "—";
        const color = KP_PLANET_COLORS[lord] ?? "#e8e3f0";

        return (
          <div key={house} className="kp-house-cell">
            <span>H{house}</span>
            <strong style={{ color }}>{lord}</strong>
          </div>
        );
      })}
    </div>
  );
}
