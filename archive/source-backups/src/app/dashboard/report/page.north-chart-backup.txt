const SIGN_ORDER = [
  "Aries",
  "Taurus",
  "Gemini",
  "Cancer",
  "Leo",
  "Virgo",
  "Libra",
  "Scorpio",
  "Sagittarius",
  "Capricorn",
  "Aquarius",
  "Pisces",
];

function getPlanetsBySign(planets: any[]) {
  const grouped: Record<string, string[]> = {};

  SIGN_ORDER.forEach((sign) => {
    grouped[sign] = [];
  });

  planets.forEach((planet) => {
    const sign = typeof planet?.sign === "string" ? planet.sign : "Unknown";
    const name = typeof planet?.name === "string" ? planet.name : "Planet";

    if (!grouped[sign]) grouped[sign] = [];
    grouped[sign].push(name);
  });

  return grouped;
}

function PremiumChartBox({
  title,
  subtitle,
  planets,
}: {
  title: string;
  subtitle: string;
  planets: any[];
}) {
  const grouped = getPlanetsBySign(planets);

  return (
    <div className="premium-chart-box">
      <div className="chart-title-row">
        <div>
          <h3>{title}</h3>
          <p>{subtitle}</p>
        </div>
        <span>AstroLife Chart</span>
      </div>

      <div className="north-chart-grid">
        {SIGN_ORDER.map((sign, index) => (
          <div key={sign} className="chart-sign-cell">
            <div className="chart-sign-top">
              <strong>{index + 1}</strong>
              <span>{sign}</span>
            </div>

            <div className="chart-planets">
              {(grouped[sign] ?? []).length ? (
                grouped[sign].map((planet) => (
                  <em key={`${sign}-${planet}`}>{planet}</em>
                ))
              ) : (
                <small>—</small>
              )}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
